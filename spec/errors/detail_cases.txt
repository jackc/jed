# Shared error-Detail cases (spec/design/error-fields.md §9.8).
#
# Every core's error_fields test runs every case and asserts the failing statement's SQLSTATE and its
# structured Detail field, byte-for-byte. The corpus cannot assert a structured field, so this one
# language-neutral file stands in for three hand-mirrored test lists (CLAUDE.md §5/§10).
#
# Format: line-oriented. A case starts at `case: <name>` and ends at the next `case:` or end of file.
# Blank lines and lines starting with `#` are ignored. Directives, in this order within a case:
#
#   setup: <sql>               run with the default (fully permissive) session; repeatable
#   default_privileges: <list> then replace the session's default table privileges; the list is
#                              comma-separated privilege names, or `-` for none
#   grant: <PRIV> <object>     then grant one privilege on one object; repeatable
#   revoke: <PRIV> <object>    then revoke one privilege on one object; repeatable
#   run: <sql>                 the statement that must fail
#   code: <sqlstate>           its SQLSTATE
#   detail: <text>             its exact Detail; the text runs to end of line (leading `detail: ` removed)
#   no-detail                  instead of `detail:`: the Detail field must be absent
#
# Expected text matches PostgreSQL 18's DETAIL wherever a PostgreSQL equivalent exists. A case marked
# `# jed:` differs from PostgreSQL by design (error-fields.md §9.7).

# --- 23505 unique ---------------------------------------------------------------------------------

case: unique primary key
setup: CREATE TABLE t (id i32 PRIMARY KEY)
setup: INSERT INTO t VALUES (1)
run: INSERT INTO t VALUES (1)
code: 23505
detail: Key (id)=(1) already exists.

case: unique quoted column label
setup: CREATE TABLE t (id i32 PRIMARY KEY, "Mixed Col" text UNIQUE)
setup: INSERT INTO t VALUES (1, 'a')
run: INSERT INTO t VALUES (2, 'a')
code: 23505
detail: Key ("Mixed Col")=(a) already exists.

case: unique composite primary key

run: INSERT INTO m VALUES (1, 'x'), (1, 'x')
code: 23505
detail: Key (a, b)=(1, x) already exists.

# jed: PostgreSQL does not truncate index key values; jed caps every value at 64 bytes.

run: INSERT INTO m VALUES (1, repeat('y', 100))
code: 23505
detail: Key (a, b)=(1, yyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyyy...) already exists.

case: unique in-statement duplicate
setup: CREATE TABLE t (id i32 PRIMARY KEY)
run: INSERT INTO t VALUES (2), (3), (2)
code: 23505
detail: Key (id)=(2) already exists.

case: unique secondary index on update

code: 23505
detail: Key (email)=(a@x) already exists.

case: unique primary key re-key on update

# --- 23503 foreign key ----------------------------------------------------------------------------

case: foreign key child insert
setup: CREATE TABLE t (id i32 PRIMARY KEY)
setup: CREATE TABLE c (id i32 PRIMARY KEY, p i32 REFERENCES t (id))
run: INSERT INTO c VALUES (1, 99)
code: 23503
detail: Key (p)=(99) is not present in table "t".

case: foreign key multi-column child insert
setup: CREATE TABLE m (a i32, b text, PRIMARY KEY (a, b))
setup: CREATE TABLE cm (id i32 PRIMARY KEY, k1 i32, k2 text, FOREIGN KEY (k1, k2) REFERENCES m (a, b))
run: INSERT INTO cm VALUES (1, 7, 'q')
code: 23503
detail: Key (k1, k2)=(7, q) is not present in table "m".

case: foreign key parent delete
setup: CREATE TABLE t (id i32 PRIMARY KEY)
setup: CREATE TABLE c (id i32 PRIMARY KEY, p i32 REFERENCES t (id))
setup: INSERT INTO t VALUES (1)
setup: INSERT INTO c VALUES (1, 1)
run: DELETE FROM t WHERE id = 1
code: 23503
detail: Key (id)=(1) is still referenced from table "c".

case: foreign key parent update
setup: CREATE TABLE t (id i32 PRIMARY KEY)
setup: CREATE TABLE c (id i32 PRIMARY KEY, p i32 REFERENCES t (id))
setup: INSERT INTO t VALUES (1)
setup: INSERT INTO c VALUES (1, 1)
run: UPDATE t SET id = 2 WHERE id = 1
code: 23503
detail: Key (id)=(1) is still referenced from table "c".

case: foreign key add constraint validation
setup: CREATE TABLE p (id i32 PRIMARY KEY)
setup: CREATE TABLE t (id i32 PRIMARY KEY, a i32)
setup: INSERT INTO t VALUES (1, 1)
run: ALTER TABLE t ADD FOREIGN KEY (a) REFERENCES p (id)
code: 23503
detail: Key (a)=(1) is not present in table "p".

case: foreign key match full mixed nulls
setup: CREATE TABLE m (a i32, b text, PRIMARY KEY (a, b))
setup: CREATE TABLE cm (id i32 PRIMARY KEY, k1 i32, k2 text, FOREIGN KEY (k1, k2) REFERENCES m MATCH FULL)
run: INSERT INTO cm VALUES (1, 7, NULL)
code: 23503
detail: MATCH FULL does not allow mixing of null and nonnull key values.

case: foreign key match full add constraint validation
setup: CREATE TABLE m (a i32, b text, PRIMARY KEY (a, b))
setup: CREATE TABLE cm (id i32 PRIMARY KEY, k1 i32, k2 text)
setup: INSERT INTO cm VALUES (1, NULL, 'q')
run: ALTER TABLE cm ADD FOREIGN KEY (k1, k2) REFERENCES m MATCH FULL
code: 23503
detail: MATCH FULL does not allow mixing of null and nonnull key values.

# The MATCH FULL sentence names no values, so the privilege gate does not degrade it (PostgreSQL
# emits it unconditionally too).
case: gate keeps foreign key match full detail without select
setup: CREATE TABLE m (a i32, b text, PRIMARY KEY (a, b))
setup: CREATE TABLE cm (id i32 PRIMARY KEY, k1 i32, k2 text, FOREIGN KEY (k1, k2) REFERENCES m MATCH FULL)
default_privileges: INSERT
run: INSERT INTO cm VALUES (1, NULL, 'q')
code: 23503
detail: MATCH FULL does not allow mixing of null and nonnull key values.

# --- 23P01 exclusion ------------------------------------------------------------------------------

case: exclusion against stored row
setup: CREATE TABLE ex (id i32 PRIMARY KEY, r i32range, EXCLUDE USING gist (r WITH &&))
setup: INSERT INTO ex VALUES (1, '[1,5)')
run: INSERT INTO ex VALUES (2, '[3,7)')
code: 23P01
detail: Key (r)=([3,7)) conflicts with existing key (r)=([1,5)).

case: exclusion within one statement
setup: CREATE TABLE ex (id i32 PRIMARY KEY, r i32range, EXCLUDE USING gist (r WITH &&))
run: INSERT INTO ex VALUES (1, '[1,5)'), (2, '[3,7)')
code: 23P01
detail: Key (r)=([3,7)) conflicts with existing key (r)=([1,5)).

case: exclusion add constraint over existing rows
setup: CREATE TABLE ex (id i32 PRIMARY KEY, r i32range)
setup: INSERT INTO ex VALUES (1, '[1,5)'), (2, '[3,7)')
run: ALTER TABLE ex ADD EXCLUDE USING gist (r WITH &&)
code: 23P01
detail: Key (r)=([3,7)) conflicts with key (r)=([1,5)).

# --- 23514 check / 23502 not null: "Failing row contains" ----------------------------------------

code: 23514
detail: Failing row contains (4, d, -1, 2020-01-01 00:00:00+00, \xdeadbeef).

case: check failing row truncates long values

run: INSERT INTO t VALUES (5, repeat('ab', 100), -1, NULL)
code: 23514
detail: Failing row contains (5, abababababababababababababababababababababababababababababababab..., -1, null).

case: check truncation keeps whole characters
setup: CREATE TABLE mb (id i32 PRIMARY KEY, s text CHECK (length(s) < 5))
run: INSERT INTO mb VALUES (1, lpad(repeat('é', 40), 41, 'a'))
code: 23514
detail: Failing row contains (1, aééééééééééééééééééééééééééééééé...).

# jed: booleans render true/false (jed's result-cell text); PostgreSQL's DETAIL shows t/f.
